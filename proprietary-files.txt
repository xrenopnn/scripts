#!/usr/bin/env bash

export BUILDER="Moto"
export HOST="ElySia"
export KDIR=$(pwd)
export OUT_DIR="${KDIR}/out"

# Android 14 Clang r510928
export COMPILER="clang"
export TOOLCHAIN_DIR="${KDIR}/toolchains"

msg() { echo -e "\n\e[1;34m[*] $1 \e[0m"; }
success() { echo -e "\n\e[1;32m[✓] $1 \e[0m"; }
warn() { echo -e "\e[1;33m[!] $1 \e[0m"; }
abort() { echo -e "\n\e[1;31m[✗] $1 \e[0m" >&2; exit 1; }

trap 'abort "Operation cancelled by user."' SIGINT

DEVICE="$1"
DEVICES=("mumba" "venice")

if [[ -z "$DEVICE" || "$DEVICE" == -* ]]; then
    abort "Usage: $0 <device> [-c|--clean] [-p|--push] [--repo=User/Repo]\nSupported devices: ${DEVICES[*]}"
fi

if [[ ! " ${DEVICES[*]} " =~ " $DEVICE " ]]; then
    abort "Unsupported device: $DEVICE"
fi

shift

CLEAN_BUILD=0
PUSH_BUILD=0
REPO_URL=""

setup_toolchain() {
    if [ ! -f "${TOOLCHAIN_DIR}/bin/${COMPILER}" ]; then
        msg "Downloading Android Clang r510928..."

        mkdir -p "${TOOLCHAIN_DIR}" || abort "Toolchain directory creation failed!"

        local URL="https://android.googlesource.com/platform/prebuilts/clang/host/linux-x86/+archive/refs/tags/android-14.0.0_r33/clang-r510928.tar.gz"

        aria2c \
            -x16 \
            -s16 \
            -k1M \
            -o "${TOOLCHAIN_DIR}/clang.tar.gz" \
            "$URL" || abort "Failed to download Android Clang!"

        tar -xf "${TOOLCHAIN_DIR}/clang.tar.gz" \
            -C "${TOOLCHAIN_DIR}" || abort "Failed to extract Android Clang!"

        rm -f "${TOOLCHAIN_DIR}/clang.tar.gz"
    fi

    export TOOLCHAIN_PATH="${TOOLCHAIN_DIR}/bin"

    export PATH="${TOOLCHAIN_PATH}:${PATH}"
    export LD_LIBRARY_PATH="${TOOLCHAIN_DIR}/lib:${LD_LIBRARY_PATH}"

    export BUILD_CC="${TOOLCHAIN_PATH}/clang"
    export CC="${BUILD_CC}"

    export ARCH=arm64
    export CROSS_COMPILE=aarch64-linux-gnu-
    export CLANG_TRIPLE=aarch64-linux-gnu-

    export LLVM=1
    export LLVM_IAS=1

    export AR="${TOOLCHAIN_PATH}/llvm-ar"
    export NM="${TOOLCHAIN_PATH}/llvm-nm"
    export LD="${TOOLCHAIN_PATH}/ld.lld"
    export STRIP="${TOOLCHAIN_PATH}/llvm-strip"
    export OBJCOPY="${TOOLCHAIN_PATH}/llvm-objcopy"
    export OBJDUMP="${TOOLCHAIN_PATH}/llvm-objdump"
    export READELF="${TOOLCHAIN_PATH}/llvm-readelf"

    export HOSTCC="${TOOLCHAIN_PATH}/clang"
    export HOSTCXX="${TOOLCHAIN_PATH}/clang++"

    KBUILD_COMPILER_STRING=$("${TOOLCHAIN_PATH}/${COMPILER}" -v 2>&1 |
        head -n 1 |
        sed 's/(https..*//' |
        sed 's/ version//')

    success "Using compiler: ${KBUILD_COMPILER_STRING}"
}

build_kernel() {
    if [[ $CLEAN_BUILD -eq 1 ]]; then
        msg "Cleaning output directory..."
        make O="${OUT_DIR}" clean >/dev/null 2>&1
        rm -rf "${OUT_DIR}"/*
    fi

    msg "Generating defconfig for ${DEVICE}..."

    make O="${OUT_DIR}" \
        KCFLAGS=-w \
        CONFIG_SECTION_MISMATCH_WARN_ONLY=y \
        gki_defconfig \
        vendor/parrot_perf.config \
        vendor/ext_config/moto-parrot.config \
        vendor/ext_config/moto-parrot-mumba.config \
        || abort "Failed to generate defconfig."

    msg "Building kernel and images..."
    BUILD_START=$(date +"%s")

    make O="${OUT_DIR}" \
        KCFLAGS=-w \
        CONFIG_SECTION_MISMATCH_WARN_ONLY=y \
        -j"$(nproc --all)" \
        || abort "Kernel compilation failed."

    msg "Installing modules..."
    rm -rf "${OUT_DIR}/modules"

    make O="${OUT_DIR}" \
        KCFLAGS=-w \
        CONFIG_SECTION_MISMATCH_WARN_ONLY=y \
        -j"$(nproc --all)" \
        INSTALL_MOD_PATH="${OUT_DIR}/modules" \
        INSTALL_MOD_STRIP=1 \
        modules_install \
        || abort "Module compilation failed."

    msg "Exporting UAPI headers..."

    make O="${OUT_DIR}" \
        KCFLAGS=-w \
        CONFIG_SECTION_MISMATCH_WARN_ONLY=y \
        -j"$(nproc --all)" \
        INSTALL_HDR_PATH="${OUT_DIR}/usr" \
        headers_install \
        || abort "Failed to export headers."

    BUILD_END=$(date +"%s")
    DIFF=$((BUILD_END - BUILD_START))

    success "Kernel built in $((DIFF / 60)) min and $((DIFF % 60)) sec."
}

export_prebuilts() {
    local PREBUILT_DIR="${OUT_DIR}/${DEVICE}-kernel-prebuilts"

    if [[ -n "$REPO_URL" ]]; then
        msg "Cloning remote prebuilts repository: ${REPO_URL}..."
        rm -rf "${PREBUILT_DIR}"

        git clone \
            "https://github.com/${REPO_URL}.git" \
            "${PREBUILT_DIR}" \
            || abort "Failed to download prebuilt kernel tree!"

        cd "${PREBUILT_DIR}" || abort "Failed to enter cloned repo."

        rm -rf dtb kernel-headers modules Image dtbo.img

        cd "${KDIR}"
    else
        msg "Structuring local prebuilts repository..."
        rm -rf "${PREBUILT_DIR}"
    fi

    mkdir -p "${PREBUILT_DIR}"/{dtb,kernel-headers/usr,modules}

    cp "${OUT_DIR}/arch/arm64/boot/Image" \
        "${PREBUILT_DIR}/Image" \
        || abort "Failed to copy Image."

    cp "${OUT_DIR}/arch/arm64/boot/dtbo.img" \
        "${PREBUILT_DIR}/dtbo.img" \
        2>/dev/null \
        || warn "dtbo.img not found."

    chmod 775 "${PREBUILT_DIR}/Image"

    find "${OUT_DIR}/arch/arm64/boot/dts" \
        -type f \
        -name "*${DEVICE}*.dtb" \
        -exec cp {} "${PREBUILT_DIR}/dtb/" \;

    find "${OUT_DIR}/modules" \
        -type f \
        -name "*.ko" \
        -exec cp {} "${PREBUILT_DIR}/modules/" \;

    cp -r "${OUT_DIR}/usr/include" \
        "${PREBUILT_DIR}/kernel-headers/usr/"

    find "${PREBUILT_DIR}/kernel-headers" \
        -type f \
        -name ".*.cmd" \
        -delete

    local K_SUBLEVEL
    K_SUBLEVEL=$(grep '^SUBLEVEL =' "${KDIR}/Makefile" | awk '{print $3}')
    [[ -z "$K_SUBLEVEL" ]] && K_SUBLEVEL="0"

    printf "VERSION = 5\nPATCHLEVEL = 4\nSUBLEVEL = %s\n\nheaders_install:\n\t@mkdir -p \$(O)/usr\n\t@rsync -mrq \$(shell pwd)/* \$(O)/\n\nall:\n\t@true\n" \
        "$K_SUBLEVEL" \
        > "${PREBUILT_DIR}/kernel-headers/Makefile"

        cd "${KDIR}"
        rm -rf "${PREBUILT_DIR}"
    else
        success "Local prebuilts successfully generated at: ${PREBUILT_DIR}"
    fi
}

setup_toolchain
build_kernel
export_prebuilts
